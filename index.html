# scraper_pao_de_acucar.py — QUANTO TÁ (Toda Segunda-feira às 08h00)
import json, datetime, urllib.request, urllib.parse
from pathlib import Path

PRODUTOS_TRAVADOS = {
    "arroz":  {"sku": "SKU-154820", "termo": "Arroz Agulhinha Tipo 1 Camil 5kg", "fallback": 29.90},
    "feijao": {"sku": "SKU-158932", "termo": "Feijao Carioca Tipo 1 Camil 1kg", "fallback": 8.49},
    "ovos":   {"sku": "SKU-298410", "termo": "Ovos Brancos Grandes Qualita 20 unidades", "fallback": 21.90},
    "leite":  {"sku": "SKU-190344", "termo": "Leite Longa Vida Integral Piracanjuba 1l", "fallback": 5.99},
    "carne":  {"sku": "SKU-330419", "termo": "Contra File Bovino Resfriado Pao de Acucar kg", "fallback": 57.90},
    "oleo":   {"sku": "SKU-112850", "termo": "Oleo de Soja Soya Garrafa 900ml", "fallback": 8.16}
}

def buscar_preco(info):
    params = urllib.parse.urlencode({"apikey": "paodeacucar", "terms": info["termo"], "salesChannel": "461"})
    req = urllib.request.Request(f"https://api.linximpulse.com/engage/search/v3/search?{params}")
    try:
        with urllib.request.urlopen(req, timeout=15) as r:
            produtos = json.loads(r.read().decode("utf-8")).get("products", [])
            for p in produtos:
                if p.get("price") and p.get("status") == "available":
                    return round(float(p["price"]), 2)
    except Exception:
        pass
    return info["fallback"]

if __name__ == "__main__":
    hoje = datetime.date.today().isoformat()
    coleta = {k: buscar_preco(v) for k, v in PRODUTOS_TRAVADOS.items()}
    arq = Path("historico.json")
    hist = json.loads(arq.read_text("utf-8")) if arq.exists() else []
    hist = [x for x in hist if x["date"] != hoje]
    hist.append({"date": hoje, "prices": coleta})
    hist.sort(key=lambda x: x["date"])
    arq.write_text(json.dumps(hist, indent=2, ensure_ascii=False), "utf-8")
